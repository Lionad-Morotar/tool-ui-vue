    2. Create `src/stories/_shared/use-story-locale.ts` with the same types and composables (so stories can import this file directly):
       ```typescript
       import { ref, computed, inject, type Ref, type ComputedRef } from 'vue'

       const storyLocaleKey = Symbol('vtu:story-locale')

       export interface StoryLocaleLabels { zh: string; en: string }

       // Module-level reactive state (shared across all stories)
       const currentLocale: Ref<string> = ref('zh-CN')

       export function useStoryLocale(labels: StoryLocaleLabels): ComputedRef<string> {
         return computed(() => currentLocale.value === 'zh-CN' ? labels.zh : labels.en)
       }

       export function useStoryLocaleRaw(): { locale: Ref<string>; toggle: () => void } {
         const ctx = inject<{ locale: Ref<string>; toggle: () => void } | null>(storyLocaleKey, null)
         if (!ctx) {
           // Fallback when not in Histoire context
           return { locale: currentLocale, toggle: () => { currentLocale.value = currentLocale.value === 'zh-CN' ? 'en' : 'zh-CN' } }
         }
         return ctx
       }
       ```
       Note: `currentLocale` is the same ref whether accessed directly or via inject, so reactivity works regardless of how stories access it.

    3. Wire the provide in histoire-setup.ts — since setupFile is loaded as a module, NOT as a Vue component, use a self-invoking approach:
       - The setupFile is executed in the Vue app context, so add a `window.__vtuStoryLocale__` object with `{ currentLocale, toggleLocale }` that the toggle component and `useStoryLocaleRaw` can both read
       - `useStoryLocale` reads from `window.__vtuStoryLocale__.currentLocale` directly (same ref, so reactivity works)

    4. Avoid: Do NOT import from `@lionad/vtu-core/i18n` — this file must have zero core dependencies (per D-03). Only use 'vue' imports.
