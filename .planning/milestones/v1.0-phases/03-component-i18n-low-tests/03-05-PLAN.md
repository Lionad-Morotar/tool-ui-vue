    describe('ComponentName i18n', () => {
      /** I18N-001: {component} renders zh-CN text by default */
      test('renders zh-CN text when wrapped in LocaleProvider with zh-CN', () => {
        const wrapper = mount(LocaleProvider, {
          props: { messages: zhCN, locale: 'zh-CN' },
          slots: { default: () => h(Component, { /* minimal props */ }) },
        });
        // Verify Chinese text is rendered (check for specific known strings)
        expect(wrapper.text()).toContain('expected Chinese text');
      });

      /** I18N-002: {component} updates text on locale switch */
      test('updates text when locale switches from zh-CN to en', async () => {
        const localeRef = ref('zh-CN');
        const wrapper = mount(LocaleProvider, {
          props: { messages: zhCN, locale: localeRef.value },
          slots: { default: () => h(Component, { /* minimal props */ }) },
        });
        expect(wrapper.text()).toContain('expected Chinese text');

        // Change locale and re-render with en messages
        await wrapper.setProps({ messages: en, locale: 'en' });
        await nextTick();
        expect(wrapper.text()).toContain('expected English text');
      });
    });
