<interfaces>
From @lionad/vtu-core/i18n:
```typescript
export function useI18n<TMessages>(): {
  t: <TKey extends DeepKeyPath<TMessages>>(key: TKey, params?: Record<string, ParamValue>) => ComputedRef<string>
  locale: ComputedRef<string>
  setLocale: (locale: string) => void
}
```

Template usage pattern:
```vue
<script setup lang="ts">
import { useI18n } from '@lionad/vtu-core/i18n'
const { t } = useI18n()
</script>

<template>
  <!-- Vue auto-unwraps ComputedRef in mustache -->
  {{ t('terminal.copied') }}
  {{ t('terminal.showAllLines', { count: lineCount }) }}
  :aria-label="isCopied ? t('terminal.copied') : t('terminal.copyOutput')"
</template>
```
